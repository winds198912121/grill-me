FORM create_sales_order.

* BAPI call
  CALL FUNCTION 'BAPI_SALESORDER_CREATEFROMDAT2'
    EXPORTING
      order_header_in = ls_order_header_in
    IMPORTING
      salesdocument   = lv_salesdocument
    TABLES
      return          = lt_return
      order_items_in  = lt_order_items_in
      order_partners  = lt_order_partners.

* Error check: loop RETURN table for type E (Error) or A (Abort)
  LOOP AT lt_return INTO ls_return
    WHERE type = 'E' OR type = 'A'.

    lv_error = abap_true.

* Log to application log (SLG1)
    CALL FUNCTION 'BAL_LOG_MSG_ADD'
      EXPORTING
        i_msgty = ls_return-type
        i_msgid = ls_return-id
        i_msgno = ls_return-number
        i_msgv1 = ls_return-message_v1
        i_msgv2 = ls_return-message_v2.

* Store in error queue for retry
    CLEAR ls_err_queue.
    ls_err_queue-errid    = ls_return-id.
    ls_err_queue-errno    = ls_return-number.
    ls_err_queue-message  = ls_return-message.
    ls_err_queue-sorg     = p_sorg.
    ls_err_queue-kunnr    = p_kunnr.
    ls_err_queue-timestamp = sy-datum && sy-uzeit.
    INSERT zsd_err_queue FROM ls_err_queue.

    EXIT.  " Exit loop on first error

* Check SY-SUBRC
  IF lv_error = abap_true.
* Rollback — discard all changes

  ENDIF.

ENDFORM.
